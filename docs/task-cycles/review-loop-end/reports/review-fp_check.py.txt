import sys, re, importlib.util
root = sys.argv[1]
def load(name, path):
    spec = importlib.util.spec_from_file_location(name, path); m = importlib.util.module_from_spec(spec); spec.loader.exec_module(m); return m
fp = load("fp", root + "/relation-model/fingerprint.py")
g = load("g", root + "/elements/decisions/format_guard.py")
text = open(root + "/foundations/decisions.md", encoding="utf-8").read()
entries = [e for e in g.parse_log(text, "WAY") if e.kind == "version"]
latest = {}
for e in entries: latest[e.number] = e
cur = {n: fp.fingerprint(e.binding) for n, e in latest.items()}
stale = 0
for e in entries:
    rel = e.fields.get("Relations", ["", 0])[0]
    for n, f in re.findall(r"WAY-(\d+)@([0-9a-f]+)", rel):
        if latest[e.number] is e and cur[int(n)] != f:
            stale += 1; print("STALE", e.number, n, f, cur[int(n)])
for n in (125, 161, 162, 163, 164): print(n, cur[n], g.count_sentences(latest[n].binding))
print("stale", stale)
