import sys, difflib
a=open(sys.argv[1],encoding='utf-8').read().split(); b=open(sys.argv[2],encoding='utf-8').read().split()
sm=difflib.SequenceMatcher(None,a,b,autojunk=False)
n=0
for op,i1,i2,j1,j2 in sm.get_opcodes():
    if op!='equal':
        n+=1; print(op,'| -',' '.join(a[i1:i2]),'| +',' '.join(b[j1:j2]))
print('changed spans:',n)
